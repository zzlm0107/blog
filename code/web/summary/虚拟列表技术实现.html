虚拟列表技术实现

正2026/7/30大约 9 分钟

虚拟列表（Virtual List）是一种优化长列表渲染性能的技术。其核心思想是按需渲染：只渲染用户当前视口内可见的少量DOM节点，对不可见的元素进行"虚拟化"处理，不占用真实DOM。当用户滚动时，动态计算并更新视口内的数据项；进而显著提升渲染性能。

主要分为元素固定高度的虚拟列表和元素不定高度的虚拟列表。 二者实现思路都是先渲染一个初始的占位元素撑起总高度，模拟完整列表的滚动效果。这个占位元素是相对定位的，列表每项是绝对定位的，然后根据当前滚动位置scrollTop, 计算起始索引startIndex和结束索引endIndex，确定当前可见的项的范围再渲染到页面上即可。

实现思路

  1. 根据长列表的数据项数量和每项的高度（分为不定高和固定高度，固定高度好处理，不定高需要动态计算），设置一个初始的占位元素撑起总高度。

  2. 滚动时获取当前滚动位置scrollTop, 然后根据scrollTop、容器高度和单项高度，计算起始索引startIndex和结束索引endIndex，确定当前可见的项的范围。

  3. 仅渲染[startIndex, endIndex]区间内的列表项，其他项进行虚拟化处理，不占用真实DOM。

  4. 监听滚动事件，滚动时依次触发第二步和第三步即可。

补充：对于每项高度不固定的虚拟滚动，需要先给每一项预估高度，渲染真实DOM后重新测算并修正，并维护一个位置记录表，存储每一项的偏移量和高度，当滚动时通过二分查找快速定位到当前可见的项的范围。

元素固定高度的虚拟列表

元素固定高度的虚拟列表，实现起来较为简单，实现思路主要是:

  1. 根据长列表的数据项数量和每项的高度, 设置一个初始的占位元素撑起总高度，模拟完整列表的滚动效果；

  2. 滚动时获取当前滚动位置scrollTop, 然后根据scrollTop、容器高度和单项高度，计算起始索引startIndex和结束索引endIndex，确定当前可见的项的范围；

  3. 仅渲染[startIndex, endIndex]区间内的列表项；

  4. 监听滚动事件，滚动时依次触发第二步和第三步即可。

优化点

  • 设置缓冲区，在可视区域前后额外渲染若干项，防止快速滚动时出现白屏；
  • 滚动事件节流：使用requestAnimationFrame或节流函数控制滚动事件触发频率，避免主线程阻塞；

下面是一个固定高度的虚拟列表实现demo：

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <style>
        #list-container {
            width: 200px;
            height: 600px;
            overflow: auto;
        }
        #list-container .list-item {
            width: 100%;
            height: 50px;
            line-height: 50px;
            text-align: center;
            color: #333;
            box-sizing: border-box;
            border: 1px solid #ccc;
        }
    </style>
</head>
<body>
    <div id="list-container" onscroll="scrollHandler()">
        <div id="placeholder-element">
            <!-- <div class="list-item"></div>
            <div class="list-item"></div>
            <div class="list-item"></div> -->
        </div>
    </div>
</body>
</html>

<script>
    /** * ------------------- 固定高度的虚拟列表实现原理 ------------------- */

    // 创建一个长列表数组
    let listData = []
    for (let i = 0; i < 100000; i++) {
        listData.push({
            id: i,
            name: `item${i}`
        })
    }
    // 数据量
    const dataLength = listData.length
    // 上下缓冲项数量
    const bufferItemCount = 5
    // 每项固定高度
    const itemHeight = 50

    // 1.根据数据量，计算占位元素的总高度，模拟完整列表的滚动效果
    function setPlaceholderElementStyle() {
        let placeholderHeight = dataLength * itemHeight
        let placeholderElement = document.getElementById('placeholder-element')
        placeholderElement.style.position = 'relative'
        placeholderElement.style.height = `${placeholderHeight}px`
    }

    // 2. 容器滚动时，根据滚动位置，计算需要渲染的项索引范围
    function getViewportItemsIndex() {
        const listContainer = document.getElementById('list-container')
        const scrollTop = listContainer.scrollTop
        const containerHeight = listContainer.clientHeight

        // 计算开始索引：向上取整，减去缓冲区
        let startIndex = Math.floor(scrollTop / itemHeight) - bufferItemCount
        // 计算结束索引：向下取整，加上缓冲区
        let endIndex = Math.ceil((scrollTop + containerHeight) / itemHeight) + bufferItemCount

        // 边界处理：确保索引在有效范围内
        startIndex = Math.max(0, startIndex)
        endIndex = Math.min(dataLength - 1, endIndex)

        return { startIndex, endIndex }
    }

    // 3. 根据索引范围渲染可视区域的项
    function renderItems(startIndex, endIndex) {
        const placeholderElement = document.getElementById('placeholder-element')
        let html = ''
        for (let i = startIndex; i <= endIndex; i++) {
            html += `
                <div class="list-item" style="position: absolute; top: ${i * itemHeight}px;">
                    ${listData[i].name}
                </div>
            `
        }
        placeholderElement.innerHTML = html
    }

    // 4. 滚动事件处理，通过 rAF 节流，保证每帧最多渲染一次
    let scrollRafId = null
    function scrollHandler() {
        if (scrollRafId) return
        scrollRafId = requestAnimationFrame(() => {
            const { startIndex, endIndex } = getViewportItemsIndex()
            renderItems(startIndex, endIndex)
            scrollRafId = null
        })
    }

    // 虚拟列表初始化设置占位高度，渲染初始可视区域的项
    setPlaceholderElementStyle()
    scrollHandler()
</script>

元素不定高度的虚拟列表

元素不定高度的虚拟列表，实现起来就复杂一些，需要维护一个列表元素高度和定位位置的记录表。然后在滚动时，由于每项的高度不固定，需要通过二分查找这个记录表，快速定位到当前可见项的索引范围，且元素渲染后同步更新这个记录表和占位元素的高度。下面是主要的实现思路：

  1. 创建位置记录表（记录每项元素的高度，定位top，通过元素索引查找），并预估一个元素平均高度，然后根据长列表的数据项数量和预估平均高度, 初始化这个位置记录表，并设置一个初始的占位元素撑起总高度，模拟完整列表的滚动效果；

  2. 滚动时获取当前滚动位置scrollTop, 然后根据scrollTop，在位置记录表中二分查找，确定第一个可见项的top位置，确定起始索引startIndex，然后在根据容器高度计算出对应的结束索引endIndex，确定当前可见的项的范围；

  3. 仅渲染[startIndex, endIndex]区间内的列表项；渲染后需测量实际高度，更新位置记录表与占位元素高度；在触发重渲染，渲染真实的可见项。

  4. 监听滚动事件，滚动时依次触发第二步和第三步即可。

优化点

  • 设置缓冲区，在可视区域前后额外渲染若干项，防止快速滚动时出现白屏；
  • 滚动事件节流：使用requestAnimationFrame或节流函数控制滚动事件触发频率，避免主线程阻塞；
  • 二分查找：在位置记录表中二分查找，确定第一个可见项的top位置；而不是遍历一遍，优化时间复杂度；
  • 渲染后需测量实际高度，如果高度有变化触发重渲染时，只需要更新第一个高度变化后面的元素位置记录即可，因为后续元素的位置是通过上一个元素的top + height 计算的，第一个高度变化的元素前面的元素位置不会受到影响。

下面是一个不定高度的虚拟列表实现demo：

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <style>
        #list-container {
            width: 200px;
            height: 600px;
            overflow: auto;
        }
        #list-container .list-item {
            width: 100%;
            text-align: center;
            color: #333;
            box-sizing: border-box;
            border: 1px solid #ccc;
        }
    </style>
</head>
<body>
    <div id="list-container" onscroll="scrollHandler()">
        <div id="placeholder-element">
            <!-- <div class="list-item"></div>
            <div class="list-item"></div>
            <div class="list-item"></div> -->
        </div>
    </div>
</body>
</html>

<script>
    /** * ------------------- 不定高度的虚拟列表实现原理 ------------------- */

    // 创建一个长列表数组
    let listData = []
    for (let i = 0; i < 100000; i++) {
        listData.push({
            id: i,
            name: `item${i}`,
            randomHeight: Math.floor(Math.random() * 120) + 30 // 随机渲染高度30-150px
        })
    }
    // 数据量
    const dataLength = listData.length
    // 上下缓冲项数量
    const bufferItemCount = 5
    // 预估的每项高度
    const itemHeight = 90
    // 每项高度与位置的记录表
    const itemHeightMap = {} // index 元素索引: {height 元素真实高度, top 元素顶部位置}

    // 初始化每项高度与位置的记录表，初始给个预估高度
    listData.forEach((item, index) => {
        itemHeightMap[index] = {
            height: itemHeight,
            top: index * itemHeight
        }
    })

    // 1.找到最后一个元素，计算占位元素的总高度
    function setPlaceholderElementStyle() {
        const lastItem = itemHeightMap[dataLength - 1];
        const placeholderHeight = lastItem.top + lastItem.height;

        let placeholderElement = document.getElementById('placeholder-element')
        placeholderElement.style.position = 'relative'
        placeholderElement.style.height = `${placeholderHeight}px`
    }

    // 2.容器滚动时，根据滚动位置，计算需要渲染的项索引范围
    function getViewportItemsIndex() {
        const listContainer = document.getElementById('list-container')
        const scrollTop = listContainer.scrollTop
        const containerHeight = listContainer.clientHeight

        // 二分查找：找到第一个 top <= scrollTop 的项
        let startIndex = binarySearchPosition(scrollTop)

        // 从起始位置开始，累加高度直到超过可视区域
        let endIndex = startIndex
        let accumulatedHeight = 0

        while (endIndex < dataLength && accumulatedHeight < containerHeight) {
            accumulatedHeight += itemHeightMap[endIndex].height
            endIndex++
        }

        // 添加缓冲区
        startIndex = Math.max(0, startIndex - bufferItemCount)
        endIndex = Math.min(dataLength - 1, endIndex + bufferItemCount)

        return { startIndex, endIndex }
    }

    // 二分查找：根据滚动位置找到对应的项索引
    function binarySearchPosition(scrollTop) {
        let left = 0
        let right = dataLength - 1

        while (left < right) {
            const mid = Math.floor((left + right) / 2)
            const midItemTop = itemHeightMap[mid].top

            if (midItemTop < scrollTop) {
                left = mid + 1
            } else {
                right = mid
            }
        }

        // 回退检查：scrollTop 可能落在前一项的内部（top < scrollTop < top + height）
        if (left > 0 && itemHeightMap[left].top > scrollTop) {
            left--
        }

        return left
    }

    // 3.渲染可视区域的项
    function renderViewportItems(startIndex, endIndex) {
        const placeholderElement = document.getElementById('placeholder-element')
        placeholderElement.innerHTML = ''

        let html = ''
        for (let i = startIndex; i <= endIndex; i++) {
            const item = listData[i]
            const positionInfo = itemHeightMap[i]

            html += `
                <div data-index="${i}" class="list-item" style="position: absolute; top: ${positionInfo.top}px; height: ${item.randomHeight}px; line-height: ${item.randomHeight}px;">
                    ${item.name}
                </div>
            `
        }
        placeholderElement.innerHTML = html
    }

    // 4. 监听滚动事件处理
    let scrollRafId = null
    function scrollHandler() {
        if (scrollRafId) return
        scrollRafId = requestAnimationFrame(() => {
            const { startIndex, endIndex } = getViewportItemsIndex()
            renderViewportItems(startIndex, endIndex)

            // 渲染后需测量实际高度进行记录更新
            updateHeightMap()
            
            scrollRafId = null
        })
    }

    // 5.渲染后，测量实际高度是否有变化，决定是否更新位置记录表与占位元素高度
    function updateHeightMap() {
        const items = document.querySelectorAll('.list-item')
        let heightChangeIndex = -1 // 渲染元素中高度变化的第一个元素的索引

        items.forEach((item) => {
            const index = item.dataset.index
            const actualHeight = item.offsetHeight

            // 如果实际高度与预估不同，更新位置记录表，只记录第一个发生变化项的索引
            if (actualHeight !== itemHeightMap[index].height) {
                itemHeightMap[index].height = actualHeight
                if (heightChangeIndex === -1) heightChangeIndex = index
            }
        })

        // 如果有高度变化，重新计算第一个变化的元素后面的所有位置，并更新占位元素高度，触发重新渲染
        if (heightChangeIndex !== -1) {
            let currentTop = itemHeightMap[heightChangeIndex].top
            for (let i = heightChangeIndex; i < dataLength; i++) {
                itemHeightMap[i].top = currentTop
                currentTop += itemHeightMap[i].height
            }
            setPlaceholderElementStyle()
            const { startIndex, endIndex } = getViewportItemsIndex()
            renderViewportItems(startIndex, endIndex)
        }
    }

    // 虚拟列表初始化渲染初始可视区域的项
    scrollHandler()
</script>
